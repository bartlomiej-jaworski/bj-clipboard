# 1. Ensure Git's official prompt helper is loaded
if ! declare -F __git_ps1 >/dev/null 2>&1; then
    GIT_EXEC_PATH="$(git --exec-path 2>/dev/null)"
    COMPLETION_PATH="${GIT_EXEC_PATH%/libexec/git-core}"
    COMPLETION_PATH="${COMPLETION_PATH%/lib/git-core}/share/git/completion"
    if [ -f "$COMPLETION_PATH/git-prompt.sh" ]; then
        . "$COMPLETION_PATH/git-completion.bash" 2>/dev/null
        . "$COMPLETION_PATH/git-prompt.sh" 2>/dev/null
    fi
fi

# 2. DO NOT set dirty/untracked/upstream checks — this keeps it instant
unset GIT_PS1_SHOWDIRTYSTATE
unset GIT_PS1_SHOWUNTRACKEDFILES
unset GIT_PS1_SHOWUPSTREAM
unset GIT_PS1_SHOWSTASHSTATE

# Powerline & Nerd Font Glyphs
ROUND_L=""
ROUND_R=""
ARROW=""
ICON_DIR=""
ICON_GIT=""
ICON_BRANCH=""
ICON_BOLT=""
ICON_OK=""
ICON_ERR=""

# Readline non-printing wrappers \001 and \002
wrap_esc() {
    printf '\001\e[%sm\002' "$1"
}

RESET=$(wrap_esc "0")
FG_YELLOW=$(wrap_esc "38;2;243;174;53")
BG_YELLOW=$(wrap_esc "48;2;243;174;53")

FG_ORANGE=$(wrap_esc "38;2;240;118;35")
BG_ORANGE=$(wrap_esc "48;2;240;118;35")

FG_BLUE=$(wrap_esc "38;2;75;149;233")
BG_BLUE=$(wrap_esc "48;2;75;149;233")

FG_DARK=$(wrap_esc "38;2;38;43;68")
FG_WHITE=$(wrap_esc "38;2;224;222;244")

FG_RED=$(wrap_esc "38;2;224;86;102")
BG_RED=$(wrap_esc "48;2;224;86;102")

build_prompt() {
    local exit_code=$?

    # 1. User Pill (alice)
    local p="${FG_YELLOW}${ROUND_L}${BG_YELLOW}${FG_DARK} \u  "

    # 2. Separator: Yellow -> Orange
    p+="${FG_YELLOW}${BG_ORANGE}${ARROW}"

    # 3. Directory Pill (path)
    p+="${FG_WHITE}  ${ICON_DIR} \w  "

    # 4. Git Segment (Native __git_ps1 with fast fallback)
    local git_branch=""
    if declare -F __git_ps1 >/dev/null 2>&1; then
        git_branch=$(__git_ps1 "%s")
    else
        git_branch=$(git branch --show-current 2>/dev/null || git rev-parse --short HEAD 2>/dev/null)
    fi

    local last_fg
    if [ -n "$git_branch" ]; then
        p+="${FG_ORANGE}${BG_YELLOW}${ARROW}"
        p+="${FG_DARK}  ${ICON_GIT} ${ICON_BRANCH}${git_branch}  ${FG_WHITE}${ICON_BOLT}  "
        last_fg="$FG_YELLOW"
    else
        last_fg="$FG_ORANGE"
    fi

    # 5. Status Pill (check / error)
    if [ $exit_code -eq 0 ]; then
        p+="${last_fg}${BG_BLUE}${ARROW}"
        p+="${FG_WHITE}  ${ICON_OK}  "
        p+="${RESET}${FG_BLUE}${ROUND_R}"
    else
        p+="${last_fg}${BG_RED}${ARROW}"
        p+="${FG_WHITE}  ${ICON_ERR}  "
        p+="${RESET}${FG_RED}${ROUND_R}"
    fi

    p+="${RESET} "
    PS1="$p"
}

PROMPT_COMMAND="build_prompt"
