__build_svg_exact_prompt() {
    local LAST_EXIT=$?
    
    # Truecolor (24-bit) ANSI Escapes matching the SVG palette
    local RESET="\[\033[0m\]"
    
    # Background Colors
    local BG_YELLOW="\[\033[48;2;243;174;53m\]"
    local BG_ORANGE="\[\033[48;2;240;118;35m\]"
    local BG_BLUE="\[\033[48;2;75;149;233m\]"
    
    # Foreground Colors
    local FG_YELLOW="\[\033[38;2;243;174;53m\]"
    local FG_ORANGE="\[\033[38;2;240;118;35m\]"
    local FG_BLUE="\[\033[38;2;75;149;233m\]"
    local FG_DARK="\[\033[38;2;38;43;68m\]"
    local FG_LIGHT="\[\033[38;2;224;222;244m\]"
    local FG_WHITE="\[\033[38;2;255;255;255m\]"

    # Segment Caps & Dividers
    local CAP_L=""
    local DIV_ARROW=""
    local CAP_R=""

    # ==========================================
    # 1. USER SEGMENT
    # ==========================================
    # Closes with a right cap and an added normal string space " "
    local SEG_USER="${FG_YELLOW}${CAP_L}${BG_YELLOW}${FG_DARK} \u ${RESET}${FG_YELLOW}${CAP_R}${RESET} "

    # ==========================================
    # 2. DIRECTORY PATH SEGMENT
    # ==========================================
    # Starts cleanly with a fresh left cap instead of an arrow extension
    local SEG_DIR="${FG_ORANGE}${CAP_L}${BG_ORANGE}${FG_LIGHT}  \w ${RESET}"

    # ==========================================
    # 3. GIT STATUS SEGMENT 
    # ==========================================
    local GIT_DATA
    GIT_DATA=$(__compute_git_metrics)
    local SEG_GIT=""
    if [ -n "$GIT_DATA" ]; then
        SEG_GIT="${FG_ORANGE}${BG_YELLOW}${DIV_ARROW}${FG_DARK}${GIT_DATA}${RESET}"
        local SEG_ROOT="${FG_YELLOW}${BG_YELLOW}${DIV_ARROW}${FG_LIGHT}  ${RESET}${FG_YELLOW}${CAP_R}${RESET} "
    else
        local SEG_ROOT="${FG_ORANGE}${BG_YELLOW}${DIV_ARROW}${FG_LIGHT}  ${RESET}${FG_YELLOW}${CAP_R}${RESET} "
    fi

    # ==========================================
    # 4. STATUS INDICATOR SEGMENT (Isolated with spaces)
    # ==========================================
    # Starts with a fresh left cap after the gap space
    local SEG_STATUS
    if [ "$LAST_EXIT" -eq 0 ]; then
        SEG_STATUS="${FG_BLUE}${CAP_L}${BG_BLUE}${FG_LIGHT}  ${RESET}${FG_BLUE}${CAP_R}${RESET}"
    else
        local BG_RED="\[\033[48;2;255;95;95m\]"
        local FG_RED="\[\033[38;2;255;95;95m\]"
        local SEG_STATUS="${FG_RED}${CAP_L}${BG_RED}${FG_WHITE}  ${RESET}${FG_RED}${CAP_R}${RESET}"
    fi

    # Assemble line 1 and line 2
    PS1="${SEG_USER}${SEG_DIR}${SEG_GIT}${SEG_ROOT}${SEG_STATUS}\n_ "
}
