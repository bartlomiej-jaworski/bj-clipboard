GIT_PS1_SHOWDIRTYSTATE=1
GIT_PS1_SHOWUNTRACKEDFILES=1
GIT_PS1_SHOWUPSTREAM="auto"

# Powerline / Nerd Font Symbols (from SVG)
ROUND_L=$'\ue0b6'      # 
ROUND_R=$'\ue0b4'      # 
ARROW=$'\ue0b0'        # 
ICON_DIR=$'\uea83'      # 
ICON_GIT=$'\uf408'      # 
ICON_BOLT=$'\uf0e7'     # 
ICON_OK=$'\uf00c'       # 
ICON_ERR=$'\uf00d'      # 

RESET='\[\e[0m\]'

# Exact palette from SVG
FG_YELLOW='\[\e[38;2;243;174;53m\]'
BG_YELLOW='\[\e[48;2;243;174;53m\]'

FG_ORANGE='\[\e[38;2;240;118;35m\]'
BG_ORANGE='\[\e[48;2;240;118;35m\]'

FG_BLUE='\[\e[38;2;75;149;233m\]'
BG_BLUE='\[\e[48;2;75;149;233m\]'

FG_DARK='\[\e[38;2;38;43;68m\]'
FG_WHITE='\[\e[38;2;224;222;244m\]'

FG_RED='\[\e[38;2;224;86;102m\]'
BG_RED='\[\e[48;2;224;86;102m\]'

build_prompt() {
    local exit_code=$?

    # 1. User Pill: " alice " with symmetric spaces
    local p="${FG_YELLOW}${ROUND_L}${BG_YELLOW}${FG_DARK} \u  "

    # 2. Separator: Yellow -> Orange
    p+="${FG_YELLOW}${BG_ORANGE}${ARROW}"

    # 3. Directory Pill: "  ~/path "
    p+="${FG_WHITE}  ${ICON_DIR} \w  "

    # 4. Optional Git Segment
    local git_info=""
    if declare -F __git_ps1 >/dev/null 2>&1; then
        git_info=$(__git_ps1 "%s")
    fi

    local last_fg
    if [ -n "$git_info" ]; then
        # Separator: Orange -> Yellow
        p+="${FG_ORANGE}${BG_YELLOW}${ARROW}"
        # Git Segment + Bolt icon with trailing space before separator
        p+="${FG_DARK}  ${ICON_GIT} ${git_info} ${FG_WHITE}${ICON_BOLT}  "
        last_fg="$FG_YELLOW"
    else
        last_fg="$FG_ORANGE"
    fi

    # 5. Status Pill: Separator -> Status Icon -> Rounded Right Cap
    if [ $exit_code -eq 0 ]; then
        p+="${last_fg}${BG_BLUE}${ARROW}"
        p+="${FG_WHITE}  ${ICON_OK}  "
        p+="${RESET}${FG_BLUE}${ROUND_R}"
    else
        p+="${last_fg}${BG_RED}${ARROW}"
        p+="${FG_WHITE}  ${ICON_ERR}  "
        p+="${RESET}${FG_RED}${ROUND_R}"
    fi

    # Trailing prompt space before cursor input
    p+="${RESET} "

    PS1="$p"
}

PROMPT_COMMAND="build_prompt"
