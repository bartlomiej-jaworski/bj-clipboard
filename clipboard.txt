__build_svg_exact_prompt() {
    local LAST_EXIT=$?
    local RESET="\[\033[0m\]"
    
    # Background Colors
    local BG_YELLOW="\[\033[48;2;243;174;53m\]"
    local BG_ORANGE="\[\033[48;2;240;118;35m\]"
    local BG_BLUE="\[\033[48;2;75;149;233m\]"
    
    # Foreground Colors
    local FG_YELLOW="\[\033[38;2;243;174;53m\]"
    local FG_ORANGE="\[\033[38;2;240;118;35m\]"
    local FG_BLUE="\[\033[38;2;75;149;233m\]"
    local FG_DARK="\[\033[38;2;38;43;68m\]"
    local FG_LIGHT="\[\033[38;2;224;222;244m\]"
    local FG_WHITE="\[\033[38;2;255;255;255m\]"

    # Arrow symbols
    local POWERLINE_ARROW=""
    # Match the arrow color to the segment it leaves behind
    local GAP_ARROW_1="${FG_YELLOW}>${RESET}"
    local GAP_ARROW_2="${FG_YELLOW}>${RESET}"

    # ==========================================
    # 1. USER SEGMENT
    # ==========================================
    # Ends with a powerline arrow, resets background, then outputs a gap arrow: > >
    local SEG_USER="${BG_YELLOW}${FG_DARK} \u ${RESET}${FG_YELLOW}${POWERLINE_ARROW}${RESET} ${GAP_ARROW_1} "

    # ==========================================
    # 2. DIRECTORY PATH SEGMENT
    # ==========================================
    local SEG_DIR="${BG_ORANGE}${FG_LIGHT}  \w ${RESET}"

    # ==========================================
    # 3. GIT STATUS SEGMENT (Bleeds out of Path)
    # ==========================================
    local GIT_DATA
    GIT_DATA=$(__compute_git_metrics)
    local SEG_GIT=""
    if [ -n "$GIT_DATA" ]; then
        SEG_GIT="${FG_ORANGE}${BG_YELLOW}${POWERLINE_ARROW}${FG_DARK}${GIT_DATA}${RESET}"
        # Ends the combined path+git block with a sharp arrow, followed by a gap arrow: > >
        local SEG_ROOT="${FG_YELLOW}${BG_YELLOW}${POWERLINE_ARROW}${FG_LIGHT}  ${RESET}${FG_YELLOW}${POWERLINE_ARROW}${RESET} ${GAP_ARROW_2} "
    else
        # Ends path block early if git data is absent, followed by a gap arrow: > >
        local SEG_ROOT="${FG_ORANGE}${BG_YELLOW}${POWERLINE_ARROW}${FG_LIGHT}  ${RESET}${FG_YELLOW}${POWERLINE_ARROW}${RESET} ${GAP_ARROW_2} "
    fi

    # ==========================================
    # 4. STATUS INDICATOR SEGMENT
    # ==========================================
    local SEG_STATUS
    if [ "$LAST_EXIT" -eq 0 ]; then
        SEG_STATUS="${BG_BLUE}${FG_LIGHT}  ${RESET}${FG_BLUE}${POWERLINE_ARROW}${RESET}"
    else
        local BG_RED="\[\033[48;2;255;95;95m\]"
        local FG_RED="\[\033[38;2;255;95;95m\]"
        local SEG_STATUS="${BG_RED}${FG_WHITE}  ${RESET}${FG_RED}${POWERLINE_ARROW}${RESET}"
    fi

    # Assemble line 1 and line 2
    PS1="${SEG_USER}${SEG_DIR}${SEG_GIT}${SEG_ROOT}${SEG_STATUS}\n_ "
}
